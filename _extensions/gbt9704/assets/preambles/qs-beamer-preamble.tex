\newlength{\progressoffset}

\setbeamertemplate{headline}{
  \begin{tikzpicture}[overlay, remember picture]
    \fill[MistyTeal!25, rounded corners=1.2pt]
      ($(current page.north west)+(\progressoffset,-0.26)$)
      rectangle ($(current page.north east)+(-\progressoffset,-0.18)$);
    \begin{scope}
      \pgfmathsetmacro{\progressratio}{\insertframenumber/\inserttotalframenumber}
      \ifdim\insertframenumber pt>\dimexpr0pt\relax
        \pgfmathsetmacro{\barlen}{\paperwidth-1.2}
        \fill[left color=BambooGreen, right color=StreamBlue, rounded corners=1.2pt,
              opacity=0.9]
          ($(current page.north west)+(\progressoffset,-0.26)$)
          rectangle
          ($(current page.north west)+({\progressoffset+\barlen*\progressratio},-0.18)$);
        \fill[StreamBlue, opacity=0.65]
          ($(current page.north west)+({\progressoffset+(\paperwidth-1.2)*\progressratio},-0.22)$)
          circle (2.5pt);
        \fill[white, opacity=0.5]
          ($(current page.north west)+({\progressoffset+(\paperwidth-1.2)*\progressratio-0.04},-0.20)$)
          circle (1pt);
      \fi
    \end{scope}
  \end{tikzpicture}
